@clearignore
@setvar! "pass" 0
if find "innkeeperSerial" ground any any 12
    @setvar! "pass" 1
endif

if "pass" = 1
    overhead "Innkeeper found"
elseif skill "Chivalry" > 80
    while not findtype 30528 ground 2091 1 1
        say "[SacredJourney"
        waitforgump 1482653814 500
        gumpresponse 4 1482653814
        overhead "Waiting for gate"
        wait 1000
    endwhile
    dclicktype 30528 ground 2091 1 1
    waitforgump 3899019871
    gumpresponse 2 3899019871
    wait 1000
elseif skill "Magery" > 50
    removelist "runebookList"
    createlist "runebookList"
    if findtypelist "runebookList" "29103|8901" backpack
        foreach "myBook" in "runebookList"
            getlabel "myBook" desc
            if "runebook: Home" in desc
                while pass = 0
                    hotkey "Recall"
                    wft 5000
                    target "myBook"
                    wait 500
                    overhead "Looking for innkeeper" 1159
                    @clearignore
                    if find "innkeeperSerial"
                        overhead "Innkeeper found" 1159
                        @setvar! pass 1
                    else
                        overhead "Not near enough to innkeeper, trying again" 1159
                    endif
                endwhile
            endif
        endfor
    else
        overhead "Runebook missing" 1159
        if findtype 3701 "backpack" as "pack"
            dclick "pack"
            wait 300
        endif
        if findtypelist "runebookList" 29103|8901 "backpack"
            loop
        endif
        stop
    endif
    if pass = 0
        overhead "Runebook named 'Home' missing" 1159
        stop
    endif
endif

say "Escort me to my room please"
waitforgump 2393832411

    overhead "Room gump not found! Trying again" 1159
    loop
elseif ingump "Enter Your Room"
    gumpresponse 4 2393832411
elseif ingump "Rent This Room"
    gumpresponse 5 2393832411
    waitforgump 2393832411
